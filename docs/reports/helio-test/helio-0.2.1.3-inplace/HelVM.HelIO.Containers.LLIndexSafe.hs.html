   44 instance ListLike full item => IndexSafe full item where
   45   indexSafe = indexSafeLL

   47 -- | Internal functions
   48 indexSafeLL :: (MonadSafe m , ListLike full item) => full -> Int -> m item
   49 indexSafeLL l i
   50   | i < 0     = liftErrorWithTupleList "LLIndexSafe.indexSafeLL: index must be >= 0" [("i" , show i)]
   51   | ll <= i   = liftErrorWithTupleList "LLIndexSafe.indexSafeLL: index must not found" [("i" , show i) , ("length l" , show ll)]
   52   | otherwise = (pure <$> index l) i
   53     where ll = length l
